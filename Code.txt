def average(nums):
    total = 0
    if not nums:
        return 0
    for i in nums:
        total += i
    return total / len(nums)

print(average([10, 20, 30, 40]))

def remove_duplicates(lst):
    seen = set()
    result = []
    for item in lst:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

    except ZeroDivisionError:
        return 0

def count_lines(filename):
    count = 0
    with open(filename, "r") as f:
        for line in f:
            if line.strip():
                count += 1
    return count

print(count_lines("sample.txt"))

int main() {
    int *arr = malloc(5 * sizeof(int));
    if (!arr) return 1;

    int *temp = realloc(arr, 10 * sizeof(int));
    if (!temp) {
        free(arr);
        return 1;
    }

    arr = temp;
    free(arr);
    return 0;
}
