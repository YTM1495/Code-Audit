def average(nums):
    total = 0
    for i in range(len(nums)):
        total += nums[i]
    return total / len(nums)

print(average([10, 20, 30, 40]))

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, n):
        if n % i == 0:
            return False
    return True

def count_lines(filename):
    f = open(filename, "r")
    lines = f.readlines()
    count = 0
    for line in lines:
        if line != "":
            count += 1
    return count
    f.close()

print(count_lines("sample.txt"))
